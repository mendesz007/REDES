#include <stdio.h>

int main() {
    int n;
    int i;
    int dispositivos[10];
    int ativos = 0;
    int inativos = 0;
    float disponibilidade;

    printf("=== RADAR DE DISPOSITIVOS DA REDE ===\n");

    printf("Quantos dispositivos deseja verificar? (1 a 10): ");
    scanf("%d", &n);

    if (n < 1 || n > 10) {
        printf("Quantidade invalida!\n");
    } else {
        for ( i = 0; i < n; i++) {
            printf("Dispositivo %d (1 = ativo, 0 = inativo): ", i + 1);
            scanf("%d", &dispositivos[i]);

            while (dispositivos[i] != 0 && dispositivos[i] != 1) {
                printf("Valor invalido! Digite 1 ou 0: ");
                scanf("%d", &dispositivos[i]);
            }

            if (dispositivos[i] == 1) {
                ativos++;
            } else {
                inativos++;
            }
        }

        disponibilidade = (ativos * 100.0) / n;

        printf("\n");
        printf("=== RESULTADO DA ANALISE ===\n");
        printf("Dispositivos ativos: %d\n", ativos);
        printf("Dispositivos inativos: %d\n", inativos);
        printf("Disponibilidade: %.2f%%\n", disponibilidade);

        if (inativos >= 2) {
            printf("ALERTA: 2 ou mais dispositivos inativos!\n");
        } else {
            printf("Status: Nenhum alerta critico.\n");
        }
    }

    return 0;
}
